    for(let mm=1;mm<=targetM;mm++){const mKey=`${year}-${String(mm).padStart(2,'0')}`;S.settings.apportion.forEach(a=>{let v=S.monthlyCosts[mKey]?.[a.id];if(v===undefined)v=a.defaultAmount;appDeduct+=Math.floor(Number(v)*(a.ratio/100));});}

    const varE=S.expenses.filter(e=>e.payDate&&e.payDate.startsWith(String(year))).reduce((s,e)=>s+Number(e.net||e.total||0),0);

    return targetM>0?Math.floor((appDeduct+varE+subT)/targetM)*12:0;

}

function calcIT(tx){

    let i=0;

    if(tx<=1950000)i=tx*.05;

    else if(tx<=3300000)i=tx*.1-97500;

    else if(tx<=6950000)i=tx*.2-427500;

    else if(tx<=9000000)i=tx*.23-636000;

    else if(tx<=18000000)i=tx*.33-1536000;

    else i=tx*.4-2796000;

    return Math.max(0,i)*1.021;

}

function getBracket(tx){

    if(tx<=1950000)return{rate:5,limit:1950000,prevLimit:0};

    if(tx<=3300000)return{rate:10,limit:3300000,prevLimit:1950000};

    if(tx<=6950000)return{rate:20,limit:6950000,prevLimit:3300000};

    if(tx<=9000000)return{rate:23,limit:9000000,prevLimit:6950000};

    if(tx<=18000000)return{rate:33,limit:18000000,prevLimit:9000000};

    return{rate:40,limit:Infinity,prevLimit:18000000};

}

function downloadApportionCSV(){

    if(mode==='single'){

        const m=document.getElementById('ap-csv-month').value;

        months=[m];

    }else{

        const from=document.getElementById('ap-csv-from').value;

        const to=document.getElementById('ap-csv-to').value;

        if(!from||!to){customAlert('開始月と終了月を選択してください','error');return;}

        if(from>to){customAlert('開始月は終了月以前にしてください','error');return;}

            const deduct=Math.floor(actual*(a.ratio/100));

            const kamoku=a.yayoiKamoku||a.name;

            const tekiyo=`令和${jYear}年${Number(m)}月分${a.name}（事業利用分${a.ratio}%）`;

            rows.push([dateStr,kamoku,deduct,'普通預金（個人用）',deduct,tekiyo,a.torihikiSaki||'',taxKbn,taxAmt]);

        });

    });

    if(!rows.length){customAlert('出力するデータがありません','info');return;}

function extractFolderId(url){

    if(!url)return null;

    // https://drive.google.com/drive/folders/FOLDER_ID

    // https://drive.google.com/drive/u/1/folders/FOLDER_ID

    const m=url.match(/\/folders\/([a-zA-Z0-9_-]+)/);

    return m?m[1]:null;

}

function showDrivePicker(accessToken){

    const df=S.settings.driveFolders||{};

    const folderUrlMap={rc:'receipt',dv:'delivery',sc:'supplies',tr:'transport'};

    const folderKey=folderUrlMap[_drivePickerTarget];

    const folderId=extractFolderId(df[folderKey]||'');

    const view=new google.picker.DocsView()

        .setIncludeFolders(true)

        .setSelectFolderEnabled(false)

        .setMimeTypes('image/jpeg,image/png,image/gif,image/webp,image/heic,image/heif,application/pdf');

    // 設定フォルダがあればそこをデフォルトに

    if(folderId){

        view.setParent(folderId);

    }

    const picker=new google.picker.PickerBuilder()

        .enableFeature(google.picker.Feature.MULTISELECT_ENABLED)

        .setOAuthToken(accessToken)

        .addView(view)

        .setCallback((data)=>drivePickerCallback(data,accessToken))

        .build();

    picker.setVisible(true);

}

async function drivePickerCallback(data,accessToken){

    if(data.action!==google.picker.Action.PICKED)return;

    const docs=data[google.picker.Response.DOCUMENTS];

    if(!docs||!docs.length)return;

    customAlert(`${docs.length}件を取得中...`,'info');

    const files=[];

    for(const doc of docs){

        const id=doc[google.picker.Document.ID];

        const name=doc[google.picker.Document.NAME]||'file';

        const mimeType=doc[google.picker.Document.MIME_TYPE]||'image/jpeg';

        try{

            const resp=await fetch(`https://www.googleapis.com/drive/v3/files/${id}?alt=media`,{headers:{Authorization:`Bearer ${accessToken}`}});

            if(!resp.ok)throw new Error('HTTP '+resp.status);

            const blob=await resp.blob();

            files.push(new File([blob],name,{type:mimeType}));

        }catch(e){console.error('Drive fetch error:',e);}

    }

    if(!files.length){customAlert('ファイルの取得に失敗しました','error');return;}

    if(_drivePickerTarget==='rc')addRcFiles(files);

    else if(_drivePickerTarget==='dv')addDvFiles(files);

    else if(_drivePickerTarget==='sc')addScFiles(files);

    else if(_drivePickerTarget==='tr')addTrFiles(files);

    customAlert(`${files.length}件をドライブから読み込みました`,'success');

}

async function fetchAI(payload,retries=3){

    const activeKey=S.settings.geminiApiKey||API_KEY;

    const isCustomKey=(activeKey&&activeKey!==API_KEY);

    const modelsToTry=isCustomKey?['gemini-2.5-flash','gemini-2.0-flash','gemini-1.5-flash-latest']:['gemini-2.5-flash-preview-09-2025'];

    let lastError=null;

    for(const modelName of modelsToTry){

        const url=`https://generativelanguage.googleapis.com/v1beta/models/${modelName}:generateContent?key=${activeKey}`;

        const delays=[1000,2000,4000];

        for(let i=0;i<=retries;i++){

            try{

                const r=await fetch(url,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(payload)});

                if(!r.ok){let errText='';try{const errJson=await r.json();errText=errJson.error?.message||`HTTP ${r.status}`;}catch(e){errText=`HTTP ${r.status}`;}if(r.status===401||r.status===403||(r.status===400&&errText.toLowerCase().includes('api key')))throw new Error('API_KEY_ERROR');if(r.status===404||errText.toLowerCase().includes('not found')||errText.toLowerCase().includes('not supported'))throw new Error('MODEL_NOT_FOUND');throw new Error(errText);}

                const data=await r.json();return data;

            }catch(e){lastError=e;if(e.message==='API_KEY_ERROR')throw e;if(e.message==='MODEL_NOT_FOUND')break;if(i===retries)break;await new Promise(r=>setTimeout(r,delays[i]));}

        }

    }

    if(lastError&&lastError.message!=='MODEL_NOT_FOUND')throw lastError;

    throw new Error('お使いのAPIキーで利用可能なAIモデルが見つかりませんでした。');

}

// 解析結果をCSV風にtextareaへ表示する

function itemsToCsvDisplay(items){

    const header='取引日,科目,取引手段,摘要,取引先,税込金額,消費税率,軽減税率対象';

    const rows=items.map(it=>{

        const title=formatTitle(it.category, it.is_reduced);

        return [

            it.date, it.category, it.payment_method||'',

            title, it.payee, it.amount_total,

            it.tax_rate, it.is_reduced?'対象':'対象外'

        ].join(',');

    });

    return [header,...rows].join('\n');

}

function registerFromItems(items, defaultCategory){

    let count=0;

    items.forEach((it,i)=>{

        const cat=it.category||defaultCategory;

        const total=Math.round(it.amount_total)||0;

        if(!total)return;

        const rate=it.tax_rate===8?8:10;

        const isReduced=it.is_reduced||rate===8;

        const amount=Math.floor(total/(1+rate/100));

        const tax=total-amount;

        const title=formatTitle(cat, isReduced);

        S.expenses.push({

            id:Date.now().toString()+i,

            category:cat,

            payDate:it.date||new Date().toISOString().slice(0,10),

            payee:it.payee||'',

            title,amount,tax,total,withholding:0,net:total

        });

        count++;

    });

    return count;

}
